```typescript
import { createDeepAgentsHandler, langGraphAdapter } from '@deepagents-nextjs/server'

// Use langGraphAdapter for LangGraph backends

  backendUrl: process.env.BACKEND_URL,
  adapter: langGraphAdapter,
})
```

### Using Authentication
```typescript
import { createDeepAgentsHandler, getCookieToken } from '@deepagents-nextjs/server'

export const POST = createDeepAgentsHandler({
  backendUrl: process.env.BACKEND_URL,
  getToken: getCookieToken('auth_token'),
})
```

### Using Custom Schemas
```typescript
import { useDeepAgentsChat, PlanSchema } from '@deepagents-nextjs/react'

const { messages } = useDeepAgentsChat<{ 'data-plan': typeof PlanSchema }>({
  sessionId: 'abc-123',
  endpoint: '/api/chat',
  schemas: { 'data-plan': PlanSchema },
})
// messages: (Message | { type: 'data-plan', data: DataPlan })[]
```

### Testing with Mock Server
```typescript
import { createMockDeepAgentsServer } from '@deepagents-nextjs/test-utils'

```

### Remix SSE Streaming
```typescript
import { useDeepAgentsChat } from '@deepagents-nextjs/remix'

export function ChatComponent() {
  const { messages, status, start } = useDeepAgentsChat('/api/chat')
  // Uses native fetch() for SSE streaming (NOT useFetcher)
  return <ChatUI messages={messages} status={status} onSubmit={start} />
}
```
