    ```typescript
      approvalId: string;
      toolCallId: string;
      toolName: string;
      input: Record<string, unknown>;
      description: string;
      createdAt: number;   // Date.now()

    ```

    **Exported functions:**
    - `registerApproval(approval: PendingApproval): void` — set in registry Map by approvalId
    - `getApproval(approvalId: string): PendingApproval | undefined` — lazy TTL check: if expiresAt < Date.now() AND status === "waiting", set status = "timeout". Always return the record (do NOT delete on TTL — caller handles cleanup).
    - `resolveApproval(approvalId: string, decision: "approve" | "reject"): void` — idempotent: if status !== "waiting", no-op. Otherwise set status to "approved" or "rejected".
    - `cleanupApproval(approvalId: string): void` — delete from registry
    - `cleanupExpiredApprovals(): number` — delete all entries where expiresAt < Date.now() AND (status !== "approved" OR status === "approved") — i.e., delete ALL entries where expiresAt < now regardless of status. Return count of removed entries.

    **[QUORUM-4] CRITICAL — cleanupExpiredApprovals must also GC stale "approved" entries:**
    The original research code skipped status="approved" entries to avoid a race where the transform hadn't drained them yet. That race is now eliminated: the transform calls `cleanupApproval()` after drain completes (see Task 2). Therefore `cleanupExpiredApprovals()` MUST also delete entries where `status === "approved" AND expiresAt < Date.now()`. These are stale entries that were approved and drained but never cleaned up (e.g., the transform was interrupted mid-drain). The implementation:
    ```typescript
    export function cleanupExpiredApprovals(): number {
      const now = Date.now();
      let cleaned = 0;
      for (const [id, approval] of registry) {
        if (approval.expiresAt < now) {
          // Remove ALL expired entries regardless of status (approved entries
          // are safe to remove here because the transform cleans up immediately
          // after drain via cleanupApproval(); any remaining approved entry
          // past TTL is an abandoned drain that will never complete)
          registry.delete(id);
          cleaned++;
        }
      }
      return cleaned;
    }
    ```

    ```typescript
    function getRegistry(): Map<string, PendingApproval> {
      const g = globalThis as unknown as {
        __deepagents_approval_registry?: Map<string, PendingApproval>;
      };
      g.__deepagents_approval_registry ??= new Map();
      return g.__deepagents_approval_registry;
    }
    ```

    Import `SseFrame` from "./accumulator".

  <action>
    Create `packages/server/src/adapters/approvalGating.ts` implementing the approval gating transform.

    **Critical pitfalls from RESEARCH.md to avoid:**
    1. Single-frame-per-call SseTransform contract: use readyQueue (same pattern as openSwe.ts) to drain buffered frames after approval unblocks N frames
    2. Drain readyQueue at the START of each transform call, before processing the new input frame
    3. seq counter must be included in the data-approval-required payload (ApprovalSchema requires it)
    4. Concurrent approval resolution: check `approval.status === "waiting"` before buffering

    **[QUORUM-2] Global pause flag — ALL frames buffered during pending approval:**
    The transform must maintain a `streamPaused` boolean (or check `pendingApprovalsByToolCallId.size > 0`) to gate ALL incoming frames — not just toolCallId-keyed ones. Text-delta frames and other non-tool frames that arrive while any approval is pending must be pushed to `globalBufferedFrames` and returned as null. They are drained after the last pending approval resolves.

    **[QUORUM-3] Rejection emits data-error frame — NOT null:**
    When `approval.status === "rejected"`, the transform must return a `data-error` frame (not discard silently). The client needs a terminal signal. After emitting the data-error frame, clear the toolCallId entry from `pendingApprovalsByToolCallId` so subsequent frames for that toolCallId pass through normally.

    **[QUORUM-4] Cleanup after drain — call cleanupApproval from transform:**
    After step 6e drains all bufferedFrames into readyQueue AND the drain is complete (readyQueue has been fully consumed), call `cleanupApproval(approvalId)`. The implementation: track drain completion by counting how many buffered frames remain. When the readyQueue becomes empty (last buffered frame was just returned), call `cleanupApproval(approvalId)` and delete from `pendingApprovalsByToolCallId`. This is safe because drain is complete at that point.

    **Interface:**
    ```typescript
    export interface ApprovalGatingConfig {
      getApprovalConfig?: (toolCall: {
        toolCallId: string;
        toolName: string;
        input: Record<string, unknown>;
      }) => { require: boolean; timeoutMs?: number } | undefined;
    }

    export function createApprovalGatingTransform(
      config: ApprovalGatingConfig
    ): SseTransform
    ```

    **Transform logic:**
    ```
    Closure state:
      pendingApprovalsByToolCallId: Map<toolCallId, approvalId>
      readyQueue: SseFrame[]
      globalBufferedFrames: SseFrame[]   // [QUORUM-2] frames buffered during global pause
      pendingDrainMeta: Map<approvalId, { totalBuffered: number, drained: number }>
      seqCounter: number (starts at 0, increment per approval frame)

    On each call:
      1. If readyQueue.length > 0:
         a. const frame = readyQueue.shift()!
         b. [QUORUM-4] Check if any approval entry just finished draining (pendingApprovalsByToolCallId
            no longer contains that approvalId after shift). Specifically: after shift, if pendingApprovalsByToolCallId
            has an approvalId whose bufferedFrames are now all in readyQueue (readyQueue.length === 0 after shift),
            call cleanupApproval(approvalId) and delete from map.
            Simpler implementation: track a `drainingApprovalId` and `remainingDrainCount` in closure.
            When remainingDrainCount reaches 0 after a shift, call cleanupApproval(drainingApprovalId).
         c. return frame

      2. Parse frame.raw: if no "data: " prefix → return frame (pass-through — non-data frames bypass buffering)
      3. If raw === "[DONE]" → return frame

      4. [QUORUM-2] If pendingApprovalsByToolCallId.size > 0 (stream is globally paused):
         - Check if the frame belongs to a pending toolCallId (step 5 below handles tool-keyed frames).
         - For frames with NO toolCallId or with a toolCallId NOT in pendingApprovalsByToolCallId
           (e.g., text-delta frames), push to globalBufferedFrames; return null.

      5. JSON.parse the data. If parse throws → return frame (pass-through non-JSON)

      6. If parsed.type === "tool-input-start":
         a. Extract toolCallId, toolName, input from parsed
         b. Call config.getApprovalConfig?.({ toolCallId, toolName, input })
         c. If result?.require is falsy → return frame (pass-through)
         d. If require: true:
            - approvalId = crypto.randomUUID()
            - createdAt = new Date().toISOString()
            - timeoutMs = result.timeoutMs ?? 300_000
            - expiresAt = new Date(Date.now() + timeoutMs).toISOString()
            - seq = seqCounter++
            - registerApproval({ approvalId, toolCallId, toolName, input, description: `Approval required for ${toolName}`, createdAt: Date.now(), expiresAt: Date.now() + timeoutMs, bufferedFrames: [frame], status: "waiting" })
            - pendingApprovalsByToolCallId.set(toolCallId, approvalId)
            - return { raw: `data: ${JSON.stringify({ type: "data-approval-required", data: { id: approvalId, seq, actionName: toolName, description: `Approval required for ${toolName}`, arguments: input, status: "waiting", createdAt, expiresAt } })}` }

      7. Extract toolCallId from parsed (check parsed.toolCallId):
         - If toolCallId exists and pendingApprovalsByToolCallId.has(toolCallId):
           a. approvalId = pendingApprovalsByToolCallId.get(toolCallId)!
           b. approval = getApproval(approvalId)
           c. If approval === undefined (cleaned up) → return frame (safety pass-through)
           d. If approval.status === "waiting" → buffer: approval.bufferedFrames.push(frame); return null
           e. If approval.status === "approved":
              - Move approval.bufferedFrames + globalBufferedFrames to readyQueue (tool-keyed frames first, then global frames)
              - Set drainingApprovalId = approvalId; remainingDrainCount = readyQueue.length
              - pendingApprovalsByToolCallId.delete(toolCallId)
              - [QUORUM-4] If readyQueue.length === 0: call cleanupApproval(approvalId) immediately
              - Else cleanupApproval will be called after last drain (step 1b)
              - return readyQueue.shift()!  (drain starts: return first buffered frame instead of input)
           f. [QUORUM-3] If approval.status === "rejected":
              - pendingApprovalsByToolCallId.delete(toolCallId)
              - Re-add globalBufferedFrames to readyQueue (unrelated frames should still be sent)
              - return { raw: `data: ${JSON.stringify({ type: "data-error", data: { code: "approval_rejected", message: "Tool execution was rejected" } })}` }
              - (Do NOT call cleanupApproval here — let cleanupExpiredApprovals() handle GC of rejected entries)
           g. If approval.status === "timeout":
              - pendingApprovalsByToolCallId.delete(toolCallId)
              - return { raw: `data: ${JSON.stringify({ type: "data-error", data: { code: "approval_timeout", message: "Tool approval expired" } })}` }

      8. Return frame (pass-through for non-tool frames not matched above)
    ```

    **Step 1b detail — cleanup-after-drain tracking:**
    ```typescript
    // In closure:
    let drainingApprovalId: string | null = null;
    let remainingDrainCount = 0;

    // At step 1a, BEFORE shift:
    if (readyQueue.length > 0) {
      const frame = readyQueue.shift()!;
      remainingDrainCount--;
      if (remainingDrainCount === 0 && drainingApprovalId !== null) {
        cleanupApproval(drainingApprovalId);
        drainingApprovalId = null;
      }
      return frame;
    }
    ```
