    Update the implementation:

    ```typescript
    import { NextRequest, NextResponse } from 'next/server';
    import { createResumableStreamContext } from 'ai';
    import { lookupStream } from './stream-registry';

    export function isStreamReconnectEnabled(): boolean {
      return process.env.ENABLE_STREAM_RECONNECT === 'true';
    }

    // Module-level singleton — survives Next.js HMR via globalThis pattern
    let streamContext: ReturnType<typeof createResumableStreamContext> | undefined;

    function getStreamContext(): ReturnType<typeof createResumableStreamContext> {
      if (!streamContext) {
        streamContext = createResumableStreamContext({
          waitUntil: (p) => {
            // In App Router edge/node runtime, waitUntil is not always available.
            // Fall through silently — the stream is still piped; cleanup is best-effort.
            if (typeof (globalThis as unknown as { waitUntil?: (p: Promise<unknown>) => void }).waitUntil === 'function') {
              (globalThis as unknown as { waitUntil: (p: Promise<unknown>) => void }).waitUntil(p);
            }
          },
        });
      }
      return streamContext;
    }

    export function createDeepAgentsResumeHandler() {
      return async function GET(
        _: NextRequest,
        { params }: { params: Promise<{ resumeId: string }> }
      ): Promise<NextResponse> {
        if (!isStreamReconnectEnabled()) {
          return new NextResponse(
            'stream reconnection disabled (set ENABLE_STREAM_RECONNECT=true)',
            { status: 503 }
          );
        }

        const { resumeId } = await params;
        const record = lookupStream(resumeId);

        if (!record || record.done) {
          return new NextResponse(null, { status: 204 });
        }

        // Active stream found — resume it via createResumableStreamContext
        const context = getStreamContext();
        const resumableStream = await context.resumeStream(record.streamId);

        if (!resumableStream) {
          // Stream context does not have this streamId — treat as completed
          return new NextResponse(null, { status: 204 });
        }

        return new NextResponse(resumableStream, {
          status: 200,
          headers: {
            'Content-Type': 'text/plain; charset=utf-8',
            'X-Accel-Buffering': 'no',
            'Cache-Control': 'no-cache',
          },
        });
      };
    }
    ```
