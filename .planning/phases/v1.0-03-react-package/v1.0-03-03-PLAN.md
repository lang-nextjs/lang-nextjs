    ```typescript
    import type { UIMessage } from 'ai'
    import { DataErrorSchema } from './schemas'
    import type { Message, AIMessage, UserMessage, ToolCallMessage, ErrorMessage } from './types'
    import { generateId } from './types'

    function isToolCallPart(part: unknown): part is {
      type: string; toolCallId: string; state: string;
      input?: unknown; output?: unknown; errorText?: string; toolName?: string
    } {
      if (typeof part !== 'object' || part === null) return false
      const p = part as { type?: unknown }
      if (typeof p.type !== 'string') return false
      return p.type === 'dynamic-tool' || p.type.startsWith('tool-')
    }

    function toolStateToStatus(state: string): 'running' | 'complete' {
      if (state === 'input-streaming' || state === 'input-available') return 'running'
      return 'complete'
    }

    function extractToolName(part: { type: string; toolName?: string }): string {
      if (part.type === 'dynamic-tool') return part.toolName ?? 'unknown'
      return part.type.slice('tool-'.length)
    }

    /**
     * Convert AI SDK UIMessage[] to typed Message[].
     *
     * Handles: user messages, assistant text parts (→ AIMessage), tool-call parts
     * (→ ToolCallMessage), data-error parts (→ ErrorMessage).
     *
     * All other data-* parts (data-plan, data-task, data-file, data-approval, etc.)
     * are console.warn + dropped — they are not in the Message union.
     *
     * @param aiMessages - The messages array from useChat.
     * @param isStreaming - True when useChat status is 'streaming' or 'submitted'.
     */
    export function partsToMessages(aiMessages: UIMessage[], isStreaming: boolean): Message[] {
      const lastAssistantIdx = aiMessages.reduce((acc, m, i) => (m.role === 'assistant' ? i : acc), -1)
      const out: Message[] = []

      aiMessages.forEach((msg, msgIdx) => {
        const isLastAssistantStreaming = isStreaming && msgIdx === lastAssistantIdx

        if (msg.role === 'user') {
          let content = ''
          for (const part of (msg.parts ?? []) as unknown[]) {
            if (typeof part === 'object' && part !== null && (part as { type?: unknown }).type === 'text') {
              const { text } = part as { text?: unknown }
              if (typeof text === 'string') content += text
            }
          }
          const user: UserMessage = { type: 'user', id: msg.id, content, timestamp: new Date() }
          out.push(user)
          return
        }

        if (msg.role !== 'assistant') return

        const parts = (msg.parts ?? []) as unknown[]
        let textBuffer = ''
        let aiBubbleSeq = 0
        let lastAiBubbleIdx = -1
        const toolSeenIds = new Map<string, number>()

        function flushText(): void {
          if (!textBuffer) return
          const bubbleId = aiBubbleSeq === 0 ? msg.id : `${msg.id}#ai-${aiBubbleSeq}`
          const ai: AIMessage = {
            type: 'ai', id: bubbleId, content: textBuffer,
            timestamp: new Date(), isStreaming: false,
          }
          lastAiBubbleIdx = out.length
          out.push(ai)
          aiBubbleSeq += 1
          textBuffer = ''
        }

        for (const rawPart of parts) {
          if (typeof rawPart !== 'object' || rawPart === null) continue
          const partType = (rawPart as { type?: unknown }).type
          if (typeof partType !== 'string') continue

          if (partType === 'text') {
            const { text } = rawPart as { text?: unknown }
            if (typeof text === 'string') textBuffer += text
          } else if (isToolCallPart(rawPart)) {
            flushText()
            const part = rawPart
            const { toolCallId } = part
            if (typeof toolCallId !== 'string' || toolCallId === '') continue
            const state = typeof part.state === 'string' ? part.state : 'input-streaming'
            const status = toolStateToStatus(state)
            const toolName = extractToolName(part)
            const args = part.input && typeof part.input === 'object'
              ? (part.input as Record<string, unknown>) : undefined
            let result: unknown
            if (state === 'output-error') result = part.errorText ?? 'Tool error'
            else if (part.output !== undefined) result = part.output
            const toolMsg: ToolCallMessage = {
              type: 'tool-call', id: toolCallId, toolName, status,
              ...(args !== undefined ? { arguments: args } : {}),
              ...(result !== undefined ? { result } : {}),
            }
            const existingIdx = toolSeenIds.get(toolCallId)
            if (existingIdx !== undefined) { out[existingIdx] = toolMsg }
            else { toolSeenIds.set(toolCallId, out.length); out.push(toolMsg) }
          } else if (partType.startsWith('data-')) {
            if (partType === 'data-error') {
              flushText()
              const envelope = rawPart as { data?: unknown }
              const parsed = DataErrorSchema.safeParse(envelope.data)
              if (parsed.success) {
                const e = parsed.data
                const err: ErrorMessage = {
                  type: 'error', id: e.id, message: e.message, retryable: e.retryable,
                }
                out.push(err)
              } else {
                console.warn('[partsToMessages] dropped invalid data-error part', parsed.error?.message)
              }
            } else {
              console.warn('[partsToMessages] dropped unsupported data-* part type:', partType)
            }
          }
        }

        flushText()

        // Apply streaming caret to the last AI bubble
        if (isLastAssistantStreaming) {
          if (lastAiBubbleIdx >= 0) {
            ;(out[lastAiBubbleIdx] as AIMessage).isStreaming = true
          } else if (!out.some((m) => m.type === 'ai')) {
            const caret: AIMessage = {
              type: 'ai', id: msg.id, content: '', timestamp: new Date(), isStreaming: true,
            }
            out.push(caret)
          }
        }

        // Edge case: assistant message with no parts at all
        if (aiBubbleSeq === 0 && !isLastAssistantStreaming && !parts.some(isToolCallPart)) {
          out.push({ type: 'ai', id: msg.id, content: '', timestamp: new Date(), isStreaming: false })
        }
      })

      return out
    }
    ```

  <action>
    Create `packages/react/src/hook.ts`. Port from `stsfront/next-app/hooks/use-chat-companion.ts`, stripping all app-specific logic:

    ```typescript
    'use client'

    import { useMemo } from 'react'
    import { useChat } from '@ai-sdk/react'
    import { DefaultChatTransport } from 'ai'
    import type { Message } from './types'
    import { partsToMessages } from './converter'

    export interface UseDeepAgentsChatOptions {
      /** Backend session ID — caller owns session lifecycle */
      sessionId: string
      /** Endpoint URL for the SSE proxy route (e.g. '/api/chat/stream') */
      endpoint: string
      /**
       * Optional token provider. Semantics mirror the server package getToken:
       * - absent or null → no Authorization header
       * - string → Authorization: Bearer {token}
       */
      getToken?: () => Promise<string | null | undefined> | string | null | undefined
    }

    export interface UseDeepAgentsChatReturn {
      messages: Message[]
      sendMessage: (text: string) => void
      status: 'idle' | 'streaming' | 'submitted' | 'error'
      error: Error | null
    }

    /**
     * useDeepAgentsChat — wraps @ai-sdk/react useChat with DefaultChatTransport
     * pre-wired for the DeepAgents SSE proxy route.
     *
     * Usage:
     *   const { messages, sendMessage, status, error } = useDeepAgentsChat({
     *     sessionId: 'abc-123',
     *     endpoint: '/api/chat/stream',
     *   })
     */
    export function useDeepAgentsChat({
      sessionId,
      endpoint,
      getToken,
    }: UseDeepAgentsChatOptions): UseDeepAgentsChatReturn {
      // Transport is stable as long as endpoint and sessionId don't change.
      // getToken is excluded from the dep array — callers should pass a stable ref.
      const transport = useMemo(
        () =>
          new DefaultChatTransport({
            api: endpoint,
            headers: async () => {
              if (!getToken) return {}
              const token = await Promise.resolve(getToken())
              return token ? { Authorization: `Bearer ${token}` } : {}
            },
            body: () => ({ sessionId }),
          }),
        // eslint-disable-next-line react-hooks/exhaustive-deps -- getToken excluded: stable ref assumed
        [endpoint, sessionId],
      )

      const {
        messages: aiMessages,
        sendMessage: aiSendMessage,
        status,
        error,
      } = useChat({ transport })

      const isStreaming = status === 'streaming' || status === 'submitted'

      const messages = useMemo(
        () => partsToMessages(aiMessages, isStreaming),
        [aiMessages, isStreaming],
      )

      const derivedStatus: UseDeepAgentsChatReturn['status'] = isStreaming
        ? (status as 'streaming' | 'submitted')
        : error
        ? 'error'
        : 'idle'

      return {
        messages,
        sendMessage: (text: string) => aiSendMessage({ text }),
        status: derivedStatus,
        error: error ?? null,
      }
    }
    ```

    import { describe, it, expect, vi, beforeEach } from 'vitest'
    import { renderHook } from '@testing-library/react'
    import { useDeepAgentsChat } from './hook'

    // Mock @ai-sdk/react to avoid real network calls
    const mockSendMessage = vi.fn()
    const mockUseChat = vi.fn()

    vi.mock('@ai-sdk/react', () => ({
      useChat: (...args: unknown[]) => mockUseChat(...args),
    }))

    vi.mock('ai', () => ({
      DefaultChatTransport: vi.fn().mockImplementation((opts: unknown) => ({ __opts: opts })),
    }))

    beforeEach(() => {
      mockSendMessage.mockReset()
      mockUseChat.mockReturnValue({
        messages: [],
        sendMessage: mockSendMessage,
        status: 'ready',
        error: null,
      })
    })

    describe('useDeepAgentsChat()', () => {
      it('returns messages, sendMessage, status, error', () => {
        const { result } = renderHook(() =>
          useDeepAgentsChat({ sessionId: 'abc', endpoint: '/api/chat/stream' })
        )
        expect(result.current.messages).toBeDefined()
        expect(result.current.sendMessage).toBeTypeOf('function')
        expect(result.current.status).toBeDefined()
        expect(result.current.error).toBeNull()
      })
      it('messages is initially an empty array', () => {
        const { result } = renderHook(() =>
          useDeepAgentsChat({ sessionId: 'abc', endpoint: '/api/chat/stream' })
        )
        expect(result.current.messages).toEqual([])
      })
      it('status is "idle" when useChat status is "ready"', () => {
        const { result } = renderHook(() =>
          useDeepAgentsChat({ sessionId: 'abc', endpoint: '/api/chat/stream' })
        )
        expect(result.current.status).toBe('idle')
      })
      it('status is "streaming" when useChat status is "streaming"', () => {
        mockUseChat.mockReturnValue({ messages: [], sendMessage: mockSendMessage, status: 'streaming', error: null })
        const { result } = renderHook(() =>
          useDeepAgentsChat({ sessionId: 'abc', endpoint: '/api/chat/stream' })
        )
        expect(result.current.status).toBe('streaming')
      })
      it('status is "error" when useChat returns an error', () => {
        mockUseChat.mockReturnValue({ messages: [], sendMessage: mockSendMessage, status: 'ready', error: new Error('fail') })
        const { result } = renderHook(() =>
          useDeepAgentsChat({ sessionId: 'abc', endpoint: '/api/chat/stream' })
        )
        expect(result.current.status).toBe('error')
        expect(result.current.error).toBeInstanceOf(Error)
      })
      it('sendMessage calls useChat sendMessage with { text }', () => {
        const { result } = renderHook(() =>
          useDeepAgentsChat({ sessionId: 'abc', endpoint: '/api/chat/stream' })
        )
        result.current.sendMessage('hello')
        expect(mockSendMessage).toHaveBeenCalledWith({ text: 'hello' })
      })
      it('messages are derived from partsToMessages(aiMessages, isStreaming)', () => {
        const fakeUserMsg = { id: '1', role: 'user', parts: [{ type: 'text', text: 'hi' }] }
        mockUseChat.mockReturnValue({ messages: [fakeUserMsg], sendMessage: mockSendMessage, status: 'ready', error: null })
        const { result } = renderHook(() =>
          useDeepAgentsChat({ sessionId: 'abc', endpoint: '/api/chat/stream' })
        )
        expect(result.current.messages).toHaveLength(1)
        expect(result.current.messages[0].type).toBe('user')
      })
      it('getToken absent → no Authorization header in transport', () => {
        const { DefaultChatTransport } = require('ai')
        renderHook(() =>
          useDeepAgentsChat({ sessionId: 'abc', endpoint: '/api/chat/stream' })
        )
        const opts = (DefaultChatTransport as ReturnType<typeof vi.fn>).mock.calls[0][0]
        // headers function should return {} when no token
        expect(typeof opts.headers).toBe('function')
      })
      it('getToken returning a string → Authorization: Bearer {token} in transport', async () => {
        const { DefaultChatTransport } = require('ai')
        ;(DefaultChatTransport as ReturnType<typeof vi.fn>).mockClear()
        renderHook(() =>
          useDeepAgentsChat({ sessionId: 'abc', endpoint: '/api/chat/stream', getToken: () => 'mytoken' })
        )
        const opts = (DefaultChatTransport as ReturnType<typeof vi.fn>).mock.calls[0][0]
        const headers = await opts.headers()
        expect(headers).toEqual({ Authorization: 'Bearer mytoken' })
      })
      it('getToken returning null → no Authorization header in transport', async () => {
        const { DefaultChatTransport } = require('ai')
        ;(DefaultChatTransport as ReturnType<typeof vi.fn>).mockClear()
        renderHook(() =>
          useDeepAgentsChat({ sessionId: 'abc', endpoint: '/api/chat/stream', getToken: () => null })
        )
        const opts = (DefaultChatTransport as ReturnType<typeof vi.fn>).mock.calls[0][0]
        const headers = await opts.headers()
        expect(headers).toEqual({})
      })
    })
