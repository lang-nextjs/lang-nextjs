<task id="v1.0-03-02-03">
  <title>Implement types.ts — Message discriminated union</title>
  <action>
    Create `packages/react/src/types.ts`. Port from `stsfront/next-app/lib/chat/message-types.ts`, keeping only the four core message types and utilities:

    ```typescript
    /**
     * Message type system for @deepagents-nextjs/react.
     * Discriminated union with explicit type literal for each variant.
     */

    export interface UserMessage {
      type: 'user'
      id: string
      content: string
      timestamp: Date
    }

    export interface AIMessage {
      type: 'ai'
      id: string
      content: string
      timestamp: Date
      isStreaming: boolean
    }

    export interface ToolCallMessage {
      type: 'tool-call'
      id: string
      toolName: string
      arguments?: Record<string, unknown>
      status: 'running' | 'complete'
      result?: unknown
    }

    export interface ErrorMessage {
      type: 'error'
      id: string
      message: string
      retryable: boolean
    }

    /**
     * Discriminated union of all message variants.
     * Discriminator: the `type` field.
     */
    export type Message = UserMessage | AIMessage | ToolCallMessage | ErrorMessage

    /** Generate a unique ID using crypto.randomUUID() */
    export function generateId(): string {
      return crypto.randomUUID()
    }

    /**
     * Exhaustive check helper for discriminated unions.
     * Place in the default case of a switch on Message.type.
     */
    export function assertNever(value: never): never {
      throw new Error(`Unhandled discriminated union member: ${JSON.stringify(value)}`)
    }
    ```

  <action>
    Create `packages/react/src/schemas.ts`. Port from `stsfront/next-app/lib/chat/ai-sdk-events.ts`. Keep: PlanSubtaskSchema, PlanSchema (as DataPlanSchema renamed), TaskSchema, FileSchema, ApprovalSchema, DataErrorSchema, parseDataPart(). Drop: DataSubAgentSchema, DataApprovalPendingSchema, DataApprovalTimeoutSchema, DataToolApprovalSchema, AiSdkDataPartSchema, KNOWN_DATA_PART_TYPES, isKnownDataPartType.

    The `parseDataPart()` function uses a simple switch on the type string (not a discriminated union) to call safeParse on the appropriate schema:

    ```typescript
    import { z } from 'zod'

    export const PlanSubtaskSchema = z.object({
      id: z.string(),
      label: z.string(),
      status: z.enum(['pending', 'in-progress', 'done']),
    })
    export type PlanSubtask = z.infer<typeof PlanSubtaskSchema>

    export const PlanSchema = z.object({
      id: z.string(),
      seq: z.number().int().nonnegative(),
      title: z.string(),
      markdown: z.string(),
      subtasks: z.array(PlanSubtaskSchema),
      updatedAt: z.string(),
    })
    export type DataPlan = z.infer<typeof PlanSchema>

    export const TaskSchema = z.object({
      id: z.string(),
      seq: z.number().int().nonnegative(),
      taskName: z.string(),
      status: z.enum(['pending', 'in-progress', 'done']),
      description: z.string().nullish(),
      groupLabel: z.string().nullish(),
    })
    export type DataTask = z.infer<typeof TaskSchema>

    export const FileSchema = z.object({
      id: z.string(),
      seq: z.number().int().nonnegative(),
      path: z.string(),
      name: z.string(),
      language: z.string().nullish(),
      size: z.number().int().nonnegative(),
      truncated: z.boolean(),
      content: z.string().nullish(),
      updatedAt: z.string(),
    })
    export type DataFile = z.infer<typeof FileSchema>

    export const ApprovalSchema = z.object({
      id: z.string(),
      seq: z.number().int().nonnegative(),
      actionName: z.string(),
      description: z.string(),
      arguments: z.record(z.string(), z.unknown()),
      status: z.enum(['waiting', 'approved', 'rejected', 'timeout']),
      createdAt: z.string(),
      expiresAt: z.string().nullish(),
    })
    export type DataApproval = z.infer<typeof ApprovalSchema>

    export const DataErrorSchema = z.object({
      id: z.string(),
      seq: z.number().int().nonnegative(),
      code: z.string(),
      message: z.string(),
      retryable: z.boolean(),
      cause: z.record(z.string(), z.unknown()).nullish(),
    })
    export type DataError = z.infer<typeof DataErrorSchema>

    export interface ParseDataPartOk {
      ok: true
      type: string
      data: unknown
    }
    export interface ParseDataPartErr {
      ok: false
      error: z.ZodError | undefined
      raw: unknown
    }
    export type ParseDataPartResult = ParseDataPartOk | ParseDataPartErr

    const SCHEMA_MAP: Record<string, z.ZodTypeAny> = {
      'data-plan': PlanSchema,
      'data-task': TaskSchema,
      'data-file': FileSchema,
      'data-approval': ApprovalSchema,
      'data-error': DataErrorSchema,
    }

    export function parseDataPart(raw: unknown): ParseDataPartResult {
      if (typeof raw !== 'object' || raw === null) return { ok: false, error: undefined, raw }
      const envelope = raw as { type?: unknown; data?: unknown }
      if (typeof envelope.type !== 'string') return { ok: false, error: undefined, raw }
      const schema = SCHEMA_MAP[envelope.type]
      if (!schema) return { ok: false, error: undefined, raw }
      const result = schema.safeParse(envelope.data)
      if (result.success) return { ok: true, type: envelope.type, data: result.data }
      return { ok: false, error: result.error, raw }
    }
    ```
