       ```

       CREATE_RESP=$(curl -s -X POST $BLAZING_URL/v1/workspace \
         -H "Authorization: Bearer $TOKEN" \
         -H "Content-Type: application/json" \
         -d '{"image": "python:3.9-slim"}')

       echo "Created workspace: $WORKSPACE_ID"

       # EXEC
       EXEC_RESP=$(curl -s -X POST $BLAZING_URL/v1/workspace/$WORKSPACE_ID/exec \
         -H "Authorization: Bearer $TOKEN" \
         -H "Content-Type: application/json" \
         -d '{"command": "echo", "args": ["hello"]}')
       echo "Exec response: $EXEC_RESP"
       # Verify stdout contains "hello"
       echo "$EXEC_RESP" | python3 -c "import sys,json; d=json.load(sys.stdin); assert 'hello' in d['stdout'], f'Expected hello in stdout, got {d[\"stdout\"]}'; print(f'PASS: stdout={d[\"stdout\"]!r}, exit_code={d[\"exit_code\"]}')"

       # DELETE
       DELETE_RESP=$(curl -s -o /dev/null -w "%{http_code}" -X DELETE $BLAZING_URL/v1/workspace/$WORKSPACE_ID \
         -H "Authorization: Bearer $TOKEN")
       echo "Delete status: $DELETE_RESP"
       # Expected: 204 (idempotent delete)
       ```
