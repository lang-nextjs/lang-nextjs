```typescript
import { defineConfig } from 'tsup';

export default defineConfig({
  entry: ['src/index.ts'],
  format: ['esm', 'cjs'],
  dts: true,
  sourcemap: true,
  clean: true,
  target: 'es2020',
  treeshake: true,
  external: ['@remix-run/node', '@remix-run/react', 'react'],
  outExtension({ format }) {
    return {
      js: format === 'esm' ? '.mjs' : '.js',
    };
  },
});
```
CRITICAL: The `external` array prevents tsup from bundling @remix-run/react (which includes React internals) — without this, consumers get two React instances.

Create `packages/remix/vitest.config.ts` with `globals: true` and `jsdom` environment:
```typescript
import { defineConfig } from 'vitest/config';

export default defineConfig({
  test: {
    globals: true,
    environment: 'jsdom',
    include: ['src/**/*.test.ts'],
  },
});
```
WHY `globals: true`: hook tests mock `global.fetch` via `vi.stubGlobal('fetch', ...)`. Without this flag, `global.fetch` assignments in Node test environment do not work correctly — tests that mock global.fetch will silently fall through or throw. This is required for Plan 03 hook tests.
WHY `environment: 'jsdom'`: Plan 03 uses `renderHook` from `@testing-library/react` which requires a DOM environment. Node environment does not support React's DOM rendering APIs.

Create `packages/remix/src/types.ts`:
```typescript
import type { ActionFunctionArgs } from '@remix-run/node';
import type { SseFrame, SseTransform } from './accumulator';

export type { SseFrame, SseTransform };

export interface SseAdapter {
  name: string;
  transforms: SseTransform[];
}

export interface RemixHandlerOptions {
  backendUrl: string;
  adapter?: SseAdapter;
  transforms?: SseTransform[];
  getToken?: (args: ActionFunctionArgs) => Promise<string | null | undefined> | string | null | undefined;
}

export interface DeepAgentsState {
  messages: unknown[];
  status: 'idle' | 'loading' | 'streaming' | 'done' | 'error';
  error: Error | null;
}

/** Return type of useDeepAgentsChat — exported so consumers can type their props */
export type DeepAgentsChatResult = DeepAgentsState & { start: () => void };
```
IMPORTANT: `getToken` receives `ActionFunctionArgs` (NOT `RequestEvent`) — Remix-specific type. Copying SvelteKit types verbatim would be wrong here.
`DeepAgentsChatResult` is the exported return type so hook.ts return type can be referenced from types.ts and consumers can use it without re-declaring the intersection.
