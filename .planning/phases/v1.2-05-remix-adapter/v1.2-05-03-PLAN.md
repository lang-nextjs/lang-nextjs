  <behavior>
Hook signature:
`export function useDeepAgentsChat(endpoint: string, options?: { sessionId?: string; body?: Record&lt;string, unknown&gt; }): DeepAgentsChatResult`

Where `DeepAgentsChatResult` is imported from `./types` and equals `DeepAgentsState & { start: () => void }`. Using the named type from types.ts (not re-declaring the intersection inline) ensures consumers can import the return type.

The hook:
1. Initializes state: `{ messages: [], status: 'idle', error: null }`
2. Uses `useRef&lt;AbortController | null&gt;(null)` for abort control
3. Uses `useEffect` return to abort on component unmount: `return () => { abortRef.current?.abort(); }`
4. Exposes `start()` function (NOT auto-start on mount):
   - Aborts any in-progress fetch: `abortRef.current?.abort()`
   - Creates new `AbortController`, stores in `abortRef.current`
   - Resets state: `setMessages([]); setStatus('loading'); setError(null)`
   - Calls native `fetch(endpoint, { method: 'POST', signal, headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ sessionId: options?.sessionId !== undefined ? options.sessionId : undefined, ...options?.body }) })`
   - IMPORTANT: Only include sessionId in body when options?.sessionId !== undefined (same contract as SvelteKit store)
   - On response: read body with `response.body!.getReader()` + `new TextDecoder()` + `new SseFrameAccumulator()`
   - Read loop: `while (true) { const { done, value } = await reader.read(); ...}`
   - On each chunk: `accumulator.push(decoder.decode(value, { stream: true }))` → iterate frames
   - For each frame starting with `'data: '`: parse JSON, `setMessages(prev => [...prev, payload])`, if first frame also `setStatus('streaming')`
   - On done: flush accumulator, process remaining frames, `setStatus('done')`
   - On catch: if AbortError → return silently; else `setError(error); setStatus('error')`
5. Returns spread of state + `{ start }`

Note on `useEffect`: Use `useEffect(() => { return () => { abortRef.current?.abort(); }; }, [])` for cleanup only. The fetch itself is triggered imperatively by `start()`, not by the effect. This avoids auto-starting on mount (Remix's imperative fetch model).

Test approach: Tests MUST use `renderHook` from `@testing-library/react`. React hooks cannot be called outside a React component — a plain async function call to `useDeepAgentsChat()` will throw "Invalid hook call". `renderHook` creates the minimal React host context needed to invoke hooks correctly. The vitest.config.ts is already set to `environment: 'jsdom'` (Plan 01) and `@testing-library/react` is already in devDependencies (Plan 01).

`makeStream` helper — defined once at the top of hook.test.ts, used by all tests that need a streaming response:
```typescript
function makeStream(...chunks: string[]): ReadableStream<Uint8Array> {
  const encoder = new TextEncoder();
  return new ReadableStream({
    start(controller) {
      for (const chunk of chunks) {
        controller.enqueue(encoder.encode(chunk));
      }
      controller.close();
    },
  });
}
```
Each `chunk` is a raw SSE string. Example for a single data frame: `makeStream('data: {"role":"assistant","content":"hi"}\n\n')`.

`makeResponse` helper:
```typescript
function makeResponse(body: ReadableStream<Uint8Array>): Response {
  return new Response(body, {
    status: 200,
    headers: { 'Content-Type': 'text/event-stream' },
  });
}
```

   ```typescript
   import { useState, useEffect, useRef } from 'react';
   import { SseFrameAccumulator } from './accumulator';
   import type { DeepAgentsState, DeepAgentsChatResult } from './types';
   
   export function useDeepAgentsChat(
     endpoint: string,
     options?: { sessionId?: string; body?: Record<string, unknown> }
   ): DeepAgentsChatResult {
     const [messages, setMessages] = useState<unknown[]>([]);
     const [status, setStatus] = useState<DeepAgentsState['status']>('idle');
     const [error, setError] = useState<Error | null>(null);
     const abortRef = useRef<AbortController | null>(null);
   
     useEffect(() => {
       return () => { abortRef.current?.abort(); };
     }, []);
   
     function start(): void {
       abortRef.current?.abort();
       const controller = new AbortController();
       abortRef.current = controller;
   
       setMessages([]);
       setStatus('loading');
       setError(null);
   
       void (async () => {
         try {
           const requestBody: Record<string, unknown> = { ...options?.body };
           if (options?.sessionId !== undefined) {
             requestBody.sessionId = options.sessionId;
           }
   
           const response = await fetch(endpoint, {
             method: 'POST',
             signal: controller.signal,
             headers: { 'Content-Type': 'application/json' },
             body: JSON.stringify(requestBody),
           });
   
           const reader = response.body!.getReader();
           const decoder = new TextDecoder();
           const accumulator = new SseFrameAccumulator();
           let firstFrame = true;
   
           while (true) {
             const { done, value } = await reader.read();
             if (done) {
               for (const raw of accumulator.flush()) {
                 if (raw.startsWith('data: ')) {
                   const payload = JSON.parse(raw.slice(6)) as unknown;
                   setMessages(prev => [...prev, payload]);
                   if (firstFrame) { setStatus('streaming'); firstFrame = false; }
                 }
               }
               setStatus('done');
               break;
             }
             for (const raw of accumulator.push(decoder.decode(value, { stream: true }))) {
               if (raw.startsWith('data: ')) {
                 const payload = JSON.parse(raw.slice(6)) as unknown;
                 setMessages(prev => [...prev, payload]);
                 if (firstFrame) { setStatus('streaming'); firstFrame = false; }
               }
             }
           }
         } catch (err) {
           if (err instanceof Error && err.name === 'AbortError') return;
           const e = err instanceof Error ? err : new Error(String(err));
           setError(e);
           setStatus('error');
         }
       })();
     }
   
     return { messages, status, error, start };
   }
   ```
