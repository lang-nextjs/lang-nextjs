Create `.github/workflows/e2e.yml`:

```yaml
name: E2E

on:
  pull_request:
    branches: [main]
  push:
    branches: [main]

concurrency:
  group: e2e-${{ github.ref }}
  cancel-in-progress: true

jobs:
  e2e-django:
    name: E2E — Django backend
    runs-on: ubuntu-latest
    # Skip on external PRs (forks don't have access to secrets)
    if: |
      github.event_name == 'push' ||
      (github.event_name == 'pull_request' && github.event.pull_request.head.repo.full_name == github.repository)
    steps:
      - name: Checkout
        uses: actions/checkout@v4

      - name: Setup pnpm
        uses: pnpm/action-setup@v4

      - name: Setup Node.js
        uses: actions/setup-node@v4
        with:
          node-version: 22
          cache: pnpm

      - name: Install dependencies
        run: pnpm install --frozen-lockfile

      - name: Install Playwright browsers
        run: pnpm exec playwright install chromium --with-deps

      - name: Build packages
        run: pnpm build

      - name: Start Django compose stack
        working-directory: apps/django-backend
        env:
          ANTHROPIC_API_KEY: ${{ secrets.ANTHROPIC_API_KEY }}
        run: docker compose up -d

      - name: Wait for Django frontend to be healthy
        run: |
          timeout 120 bash -c 'until curl -sf http://localhost:3001/; do sleep 5; done'
          echo "Frontend ready on :3001"

      - name: Run E2E suite against Django
        env:
          PLAYWRIGHT_BASE_URL: http://localhost:3001
        run: pnpm e2e

      - name: Upload Playwright report
        if: failure()
        uses: actions/upload-artifact@v4
        with:
          name: playwright-report-django
          path: playwright-report/
          retention-days: 7

      - name: Stop Django compose stack
        if: always()
        working-directory: apps/django-backend
        run: docker compose down -v

  e2e-fastapi:
    name: E2E — FastAPI backend
    runs-on: ubuntu-latest
    # Skip on external PRs (forks don't have access to secrets)
    if: |
      github.event_name == 'push' ||
      (github.event_name == 'pull_request' && github.event.pull_request.head.repo.full_name == github.repository)
    steps:
      - name: Checkout
        uses: actions/checkout@v4

      - name: Setup pnpm
        uses: pnpm/action-setup@v4

      - name: Setup Node.js
        uses: actions/setup-node@v4
        with:
          node-version: 22
          cache: pnpm

      - name: Install dependencies
        run: pnpm install --frozen-lockfile

      - name: Install Playwright browsers
        run: pnpm exec playwright install chromium --with-deps

      - name: Build packages
        run: pnpm build

      - name: Start FastAPI compose stack
        working-directory: apps/fastapi-backend
        env:
          ANTHROPIC_API_KEY: ${{ secrets.ANTHROPIC_API_KEY }}
        run: docker compose up -d

      - name: Wait for FastAPI frontend to be healthy
        run: |
          timeout 120 bash -c 'until curl -sf http://localhost:3002/; do sleep 5; done'
          echo "Frontend ready on :3002"

      - name: Run E2E suite against FastAPI
        env:
          PLAYWRIGHT_BASE_URL: http://localhost:3002
        run: pnpm e2e

      - name: Upload Playwright report
        if: failure()
        uses: actions/upload-artifact@v4
        with:
          name: playwright-report-fastapi
          path: playwright-report/
          retention-days: 7

      - name: Stop FastAPI compose stack
        if: always()
        working-directory: apps/fastapi-backend
        run: docker compose down -v
```

Key decisions (per CONTEXT.md Claude's Discretion):
- External PR skip condition: `github.event.pull_request.head.repo.full_name == github.repository` — this is the standard GitHub Actions pattern for trusted-contributor-only jobs without using `environment:` protection rules (which require paid plans for private repos).
- `timeout 120` in wait-for-health: gives 2 minutes for Docker images to pull and containers to start. Real LLM model loading may be slow.
- `docker compose down -v` on `always()`: cleans up volumes even on test failure to avoid disk pressure between CI runs.
- `pnpm exec playwright install chromium --with-deps` installs only Chromium (matches the single project in `playwright.config.ts`) + OS dependencies.
- `pnpm build` before E2E: ensures packages/server and packages/react are built so `apps/example/` can import workspace packages.
- Jobs run independently (`e2e-django` and `e2e-fastapi` are sibling jobs, not sequential) — a Django failure does not cancel FastAPI.
