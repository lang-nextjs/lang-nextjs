`main.py` — full FastAPI app using lifespan for graph initialization:

```python
import json
import os
from uuid import uuid4
from contextlib import asynccontextmanager
from fastapi import FastAPI, Request
from fastapi.responses import StreamingResponse
from fastapi.middleware.cors import CORSMiddleware
from dotenv import load_dotenv

load_dotenv(".env.local")

# Graph is initialized at startup and reused across requests
_graph = None

@asynccontextmanager
async def lifespan(app: FastAPI):
    global _graph
    from deepagents import create_deep_agent
    _graph = create_deep_agent()
    yield
    _graph = None

app = FastAPI(lifespan=lifespan)

app.add_middleware(
    CORSMiddleware,
    allow_origins=[
        "http://localhost:3001",
        "http://localhost:3002",
        "http://frontend:3001",
        "http://frontend:3002",
    ],
    allow_methods=["POST", "OPTIONS"],
    allow_headers=["Content-Type", "Authorization"],
)

async def langgraph_to_sse(messages):
    """Convert LangGraph astream_events to DeepAgents SSE wire format."""
    text_id = "text-1"
    yield f'data: {{"type":"text-start","id":"{text_id}"}}\n\n'
    async for event in _graph.astream_events({"messages": messages}, version="v2"):
        if event["event"] == "on_chat_model_stream":
            delta = event["data"]["chunk"].content
            if delta:
                escaped = json.dumps(delta)
                yield f'data: {{"type":"text-delta","id":"{text_id}","delta":{escaped}}}\n\n'
    yield f'data: {{"type":"text-end","id":"{text_id}"}}\n\n'
    msg_id = str(uuid4())
    yield f'data: {{"type":"finish","finishReason":"stop","usage":{{"inputTokens":0,"outputTokens":0}},"messageId":"msg-{msg_id}"}}\n\n'

@app.get("/health")
async def health():
    return {"status": "ok"}

@app.post("/api/chat/stream")
async def chat_stream(request: Request):
    body = await request.json()
    messages = body.get("messages", [])
    user_text = messages[-1].get("content", "") if messages else ""
    input_messages = [{"role": "user", "content": user_text}]
    return StreamingResponse(
        langgraph_to_sse(input_messages),
        media_type="text/event-stream",
        headers={
            "Cache-Control": "no-cache",
            "X-Accel-Buffering": "no",
        },
    )
```

`docker-compose.yml` — two services only (no postgres, no redis):
```yaml
services:
  backend:
    build: .
    ports:
      - "8001:8001"
    environment:
      ANTHROPIC_API_KEY: ${ANTHROPIC_API_KEY}
    env_file:
      - path: .env.local
        required: false
    healthcheck:
      test: ["CMD-SHELL", "curl -f http://localhost:8001/health || exit 1"]
      interval: 10s
      timeout: 5s
      retries: 5
      start_period: 15s

  frontend:
    image: node:22-alpine
    working_dir: /app
    volumes:
      - ../../:/monorepo
    command: sh -c "cd /monorepo && npm install -g pnpm && pnpm install --frozen-lockfile && cd apps/example && pnpm build && pnpm start"
    ports:
      - "3002:3000"
    environment:
      BACKEND_URL: http://backend:8001
    depends_on:
      backend:
        condition: service_healthy
    healthcheck:
      test: ["CMD-SHELL", "curl -f http://localhost:3000/ || exit 1"]
      interval: 10s
      timeout: 5s
      retries: 10
      start_period: 60s
```

Note: The frontend service runs `pnpm start` (production server) after `pnpm build` to keep the container stateless and deterministic. The monorepo root is bind-mounted so the example app can access workspace packages.
