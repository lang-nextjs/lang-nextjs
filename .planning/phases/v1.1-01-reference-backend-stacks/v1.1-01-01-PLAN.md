`deepagents_backend/views.py` — async streaming view:

```python
import json
import asyncio
from uuid import uuid4
from django.http import StreamingHttpResponse
from django.views.decorators.csrf import csrf_exempt
from django.views.decorators.http import require_http_methods
from deepagents import create_deep_agent

def _get_graph():
    """Initialize the DeepAgents graph (called lazily)."""
    return create_deep_agent()

async def _langgraph_to_sse(messages):
    """Convert LangGraph astream_events to DeepAgents SSE wire format."""
    graph = _get_graph()
    text_id = "text-1"
    yield f'data: {{"type":"text-start","id":"{text_id}"}}\n\n'
    async for event in graph.astream_events({"messages": messages}, version="v2"):
        if event["event"] == "on_chat_model_stream":
            delta = event["data"]["chunk"].content
            if delta:
                escaped = json.dumps(delta)
                yield f'data: {{"type":"text-delta","id":"{text_id}","delta":{escaped}}}\n\n'
    yield f'data: {{"type":"text-end","id":"{text_id}"}}\n\n'
    msg_id = str(uuid4())
    yield f'data: {{"type":"finish","finishReason":"stop","usage":{{"inputTokens":0,"outputTokens":0}},"messageId":"msg-{msg_id}"}}\n\n'

@csrf_exempt
@require_http_methods(["POST"])
async def chat_stream(request):
    import json as _json
    body = _json.loads(request.body)
    messages = body.get("messages", [])
    user_text = messages[-1].get("content", "") if messages else ""
    input_messages = [{"role": "user", "content": user_text}]
    response = StreamingHttpResponse(
        _langgraph_to_sse(input_messages),
        content_type="text/event-stream",
    )
    response["Cache-Control"] = "no-cache"
    response["X-Accel-Buffering"] = "no"
    return response
```
