  <title>Replace placeholder page.tsx with full chat UI</title>
  <action>
    Replace `apps/example/app/page.tsx` with:

    ```tsx
    'use client'

    import { useState, useRef, useEffect } from 'react'
    import { useDeepAgentsChat } from '@deepagents-nextjs/react'
    import type { AIMessage, ToolCallMessage, UserMessage } from '@deepagents-nextjs/react'

    function UserBubble({ msg }: { msg: UserMessage }) {
      return (
        <div className="flex justify-end">
          <div className="max-w-xs rounded-2xl bg-blue-600 px-4 py-2 text-white text-sm">
            {msg.content}
          </div>
        </div>
      )
    }

    function AIBubble({ msg }: { msg: AIMessage }) {
      return (
        <div className="flex justify-start">
          <div className="max-w-sm rounded-2xl bg-white border border-gray-200 px-4 py-2 text-gray-900 text-sm shadow-sm">
            {msg.content}
            {msg.isStreaming && (
              <span className="ml-0.5 inline-block w-2 h-4 bg-gray-400 animate-pulse" />
            )}
          </div>
        </div>
      )
    }

    function ToolCallCard({ msg }: { msg: ToolCallMessage }) {
      return (
        <div className="flex justify-start">
          <div className="max-w-sm rounded-xl bg-amber-50 border border-amber-200 px-4 py-2 text-sm">
            <div className="flex items-center gap-2">
              <span className="font-mono text-amber-800">{msg.toolName}</span>
              <span
                className={
                  msg.status === 'complete'
                    ? 'rounded-full bg-green-100 text-green-800 px-2 py-0.5 text-xs'
                    : 'rounded-full bg-amber-100 text-amber-800 px-2 py-0.5 text-xs'
                }
              >
                {msg.status}
              </span>
            </div>
          </div>
        </div>
      )
    }

    export default function ChatPage() {
      const [input, setInput] = useState('')
      const bottomRef = useRef<HTMLDivElement>(null)

      const { messages, sendMessage, status } = useDeepAgentsChat({
        sessionId: 'example-session',
        endpoint: '/api/chat/stream',
      })

      // Auto-scroll to bottom on new messages
      useEffect(() => {
        bottomRef.current?.scrollIntoView({ behavior: 'smooth' })
      }, [messages])

      function handleSubmit(e: React.FormEvent) {
        e.preventDefault()
        const text = input.trim()
        if (!text || status !== 'idle') return
        sendMessage(text)
        setInput('')
      }

      return (
        <main className="flex flex-col h-screen max-w-2xl mx-auto">
          {/* Header */}
          <header className="border-b border-gray-200 bg-white px-4 py-3 flex items-center gap-2">
            <div className="h-2 w-2 rounded-full bg-green-500" />
            <span className="font-semibold text-sm">DeepAgents Example</span>
            <span className="ml-auto text-xs text-gray-400">{status}</span>
          </header>

          {/* Message list */}
          <div className="flex-1 overflow-y-auto p-4 space-y-3">
            {messages.length === 0 && (
              <div className="text-center text-gray-400 text-sm mt-8">
                Send a message to start the demo
              </div>
            )}
            {messages.map((msg) => {
              if (msg.type === 'user') return <UserBubble key={msg.id} msg={msg} />
              if (msg.type === 'ai') return <AIBubble key={msg.id} msg={msg} />
              if (msg.type === 'tool-call') return <ToolCallCard key={msg.id} msg={msg} />
              return null
            })}
            <div ref={bottomRef} />
          </div>

          {/* Input bar */}
          <form
            onSubmit={handleSubmit}
            className="border-t border-gray-200 bg-white p-4 flex gap-2"
          >
            <input
              type="text"
              value={input}
              onChange={(e) => setInput(e.target.value)}
              placeholder="Type a message…"
              disabled={status !== 'idle'}
              className="flex-1 rounded-xl border border-gray-300 px-4 py-2 text-sm outline-none focus:border-blue-400 disabled:opacity-50"
            />
            <button
              type="submit"
              disabled={status !== 'idle' || !input.trim()}
              className="rounded-xl bg-blue-600 px-4 py-2 text-sm text-white font-medium disabled:opacity-50 hover:bg-blue-700 transition-colors"
            >
              Send
            </button>
          </form>
        </main>
      )
    }
    ```

    Key implementation notes:
    - `'use client'` as first line (required for hooks)
    - Switches on `msg.type` — no UIMessage types visible in component code (EX-01 SC-2)
    - `msg.isStreaming` drives the streaming cursor in AIBubble
    - ToolCallCard shows `msg.toolName` and `msg.status`
    - `status !== 'idle'` disables the input while streaming/submitted
