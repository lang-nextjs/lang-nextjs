# Plan v1.0-04-02: Wave 2 — Mock SSE Route

## Goal

Implement `apps/example/app/api/chat/stream/route.ts` — a Next.js App Router POST handler that uses `streamText` + `MockLanguageModelV3` + `toUIMessageStreamResponse()` to produce valid AI SDK v6 UIMessageStream output. This route is what `useDeepAgentsChat()` connects to via `DefaultChatTransport`.

## Context

- AI SDK v6 `streamText().toUIMessageStreamResponse()` automatically sets:
  - `Content-Type: text/event-stream; charset=utf-8`
  - `x-vercel-ai-ui-message-stream: v1` header
  - Chunked transfer encoding (progressive delivery)
- `MockLanguageModelV3` + `simulateReadableStream` from `ai` provide the scripted response
- The mock emits: text chunks (→ AIMessage) + a tool-call sequence (→ ToolCallMessage)
- `chunkDelayInMs: 50` proves non-buffering when verified with `curl -N`
- The route ignores `sessionId` from the request body (mock always returns same response)
- `@deepagents-nextjs/server` is NOT used here — this is the backend, not a proxy

    ```typescript
    /**
     * Mock SSE route for the DeepAgents example app.
     *
     * Uses AI SDK v6 streamText + MockLanguageModelV3 to produce a scripted
     * UIMessageStream response — the same format useDeepAgentsChat() consumes.
     *
     * This route is a reference implementation showing how to wire up a real
     * DeepAgents backend: replace MockLanguageModelV3 with your actual model
     * and connect via createDeepAgentsHandler from @deepagents-nextjs/server.
     */
    import { streamText, simulateReadableStream } from 'ai'
    import { MockLanguageModelV3 } from 'ai/test'

    export const dynamic = 'force-dynamic'

    export async function POST() {
      const result = streamText({
        model: new MockLanguageModelV3({
          doStream: async () => ({
            stream: simulateReadableStream({
              // 50ms delay between chunks — proves non-buffering with curl -N
              chunkDelayInMs: 50,
              chunks: [
                // Text response — becomes AIMessage via useDeepAgentsChat
                { type: 'text-start', id: 'text-1' },
                { type: 'text-delta', id: 'text-1', textDelta: 'Hello! ' },
                { type: 'text-delta', id: 'text-1', textDelta: 'I am the mock ' },
                { type: 'text-delta', id: 'text-1', textDelta: 'DeepAgents assistant.' },
                { type: 'text-end', id: 'text-1' },
                // Tool call — becomes ToolCallMessage via useDeepAgentsChat
                {
                  type: 'tool-call',
                  toolCallId: 'tc-001',
                  toolName: 'lookup',
                  input: { query: 'example tool call' },
                },
                {
                  type: 'tool-result',
                  toolCallId: 'tc-001',
                  result: { answer: 'This is a mocked tool result.' },
                },
                // Second text after tool call — another AIMessage
                { type: 'text-start', id: 'text-2' },
                { type: 'text-delta', id: 'text-2', textDelta: 'Tool call complete. ' },
                { type: 'text-delta', id: 'text-2', textDelta: 'How can I help further?' },
                { type: 'text-end', id: 'text-2' },
                {
                  type: 'finish',
                  finishReason: 'stop',
                  usage: { promptTokens: 0, completionTokens: 0 },
                },
              ],
            }),
            rawCall: { rawPrompt: null, rawSettings: {} },
          }),
        }),
        prompt: 'mock — input is ignored',
        tools: {
          lookup: {
            description: 'Mock lookup tool',
            parameters: {
              type: 'object',
              properties: { query: { type: 'string' } },
              required: ['query'],
              additionalProperties: false,
            },
            execute: async ({ query }: { query: string }) => ({ answer: `Mock result for: ${query}` }),
          },
        },
      })

      return result.toUIMessageStreamResponse()
    }
    ```

    **Critical notes:**
    - `export const dynamic = 'force-dynamic'` prevents Next.js from caching the route response
    - The `tools` option is required for tool-call parts to be included in the UIMessageStream output
    - `toUIMessageStreamResponse()` sets `x-vercel-ai-ui-message-stream: v1` automatically
