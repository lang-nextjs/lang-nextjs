<ai_sdk_v6_wire_format>
AI SDK v6 frame types that adapters must produce. The adapter output MUST match these shapes exactly — AI SDK v6 rejects extra or missing required fields.

| Frame type          | Required fields                                          | Optional fields          |
|---------------------|----------------------------------------------------------|--------------------------|
| text-delta          | type, delta (string)                                     | id                       |
| finish              | type, finishReason ("stop" | "length" | "tool-calls")   | usage                    |
| tool-input-start    | type, toolCallId (string), toolName (string)             | —                        |
| tool-input-delta    | type, toolCallId (string), delta (string, JSON partial)  | —                        |
| tool-input-available| type, toolCallId (string), toolName (string), input (object) | —                   |
| tool-output-available| type, toolCallId (string), output (string)             | —                        |
| data-*              | type (must start with "data-"), data (any)               | —                        |

Note: `id` on text-delta is optional — do not include it unless the LangGraph frame provides a stable run_id to use.
Frames with unrecognized `type` values are silently dropped by AI SDK v6 clients; prefer explicit null returns over unknown type strings.
</ai_sdk_v6_wire_format>

    ```typescript
    // packages/server/src/adapters/langgraph.ts
    import type { SseFrame, SseTransform } from '../accumulator'
    import type { SseAdapter } from './deepagents'

    type LangGraphEvent = {
      type: string
      ns: string[]
      data: Record<string, unknown>
    }

    function langGraphToAiSdk(frame: SseFrame): SseFrame | null {
      const line = frame.raw
      if (!line.startsWith('data: ')) return frame

      const raw = line.slice(6)
      if (raw === '[DONE]') return frame

      let parsed: LangGraphEvent
      try {
        parsed = JSON.parse(raw) as LangGraphEvent
      } catch {
        return frame  // Non-JSON — pass through
      }

      if (!parsed.type) return frame  // Not a LangGraph event shape

      switch (parsed.type) {
        case 'on_llm_stream': {
          const chunk = (parsed.data?.chunk as Record<string, unknown>) ?? {}
          const content = (chunk.content as string) ?? ''
          if (!content) return null  // Empty delta — drop
          // text-delta: required fields are type + delta (AI SDK v6)
          return {
            raw: `data: ${JSON.stringify({ type: 'text-delta', delta: content })}`,
          }
        }
        case 'on_tool_start': {
          const toolName = (parsed.data?.name as string) ?? ''
          const toolId = (parsed.data?.run_id as string) ?? ''
          // tool-input-start: required fields are type + toolCallId + toolName (AI SDK v6)
          return {
            raw: `data: ${JSON.stringify({ type: 'tool-input-start', toolCallId: toolId, toolName })}`,
          }
        }
        case 'on_tool_end': {
          // IMPORTANT: This case is only present if the fixture file contains on_tool_end events.
          // If fixture validation in RED phase shows no on_tool_end frames, remove this case entirely.
          const output = (parsed.data?.output as string) ?? ''
          const toolId = (parsed.data?.run_id as string) ?? ''
          // tool-output-available: required fields are type + toolCallId + output (AI SDK v6)
          return {
            raw: `data: ${JSON.stringify({ type: 'tool-output-available', toolCallId: toolId, output })}`,
          }
        }
        case 'on_custom_event': {
          // Forward as data-* event using the custom event name
          const name = (parsed.data?.name as string) ?? 'custom'
          // data-* frames: type must start with "data-", data field is any (AI SDK v6)
          return {
            raw: `data: ${JSON.stringify({ type: `data-${name}`, data: parsed.data?.data ?? null })}`,
          }
        }
        // Drop all other events (on_llm_start, on_llm_end, on_chain_start, on_chain_end, debug)
        default:
          return null
      }
    }

    export const langGraphAdapter: SseAdapter = {
      name: 'langgraph',
      transforms: [langGraphToAiSdk],
    } as const
    ```
